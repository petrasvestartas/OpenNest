cmake_minimum_required(VERSION 3.20)
project(nest_constructive VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(Threads REQUIRED)
set(GEOMETRY_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../opennest_cpp")
add_library(constructive_core STATIC geometry.cpp engine.cpp
  "${GEOMETRY_VENDOR}/src/clipper2/clipper.engine.cpp"
  "${GEOMETRY_VENDOR}/src/clipper2/clipper.offset.cpp")
target_include_directories(constructive_core PUBLIC include PRIVATE
  "${GEOMETRY_VENDOR}/src" "${GEOMETRY_VENDOR}/third_party/boost_min")
target_link_libraries(constructive_core PUBLIC Threads::Threads)
add_library(nest_constructive SHARED capi.cpp)
target_compile_definitions(nest_constructive PRIVATE NC_BUILD)
target_link_libraries(nest_constructive PRIVATE constructive_core)
set_target_properties(nest_constructive PROPERTIES PREFIX "")
if(MSVC)
  target_compile_options(constructive_core PRIVATE /W4 /permissive- /bigobj)
  target_compile_options(nest_constructive PRIVATE /W4 /permissive-)
else()
  target_compile_options(constructive_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(nest_constructive PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MINGW)
  target_link_options(nest_constructive PRIVATE -static -static-libgcc -static-libstdc++)
endif()
include(CTest)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/bench.cpp")
  add_executable(constructive_bench bench.cpp)
  target_link_libraries(constructive_bench PRIVATE nest_constructive)
  target_include_directories(constructive_bench PRIVATE include)
endif()
if(BUILD_TESTING)
  add_executable(constructive_capi_tests tests/capi_tests.cpp)
  target_link_libraries(constructive_capi_tests PRIVATE nest_constructive)
  target_include_directories(constructive_capi_tests PRIVATE include)
  add_test(NAME constructive_capi COMMAND constructive_capi_tests)
  foreach(kind geometry search)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${kind}_tests.cpp")
      add_executable(constructive_${kind}_tests "tests/${kind}_tests.cpp")
      target_link_libraries(constructive_${kind}_tests PRIVATE constructive_core)
      add_test(NAME constructive_${kind} COMMAND constructive_${kind}_tests)
    endif()
  endforeach()
endif()
